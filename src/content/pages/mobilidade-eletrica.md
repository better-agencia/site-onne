  (function () {
    var ids = ["calc-energia", "calc-consumo", "calc-gasolina", "calc-moto"];
    var outE = document.getElementById("calc-result-energy");
    var outG = document.getElementById("calc-result-gas");
    if (!outE || !outG) return;
    function money(n) {
      if (!isFinite(n) || n < 0) return "—";
      return "R$ " + n.toFixed(2).replace(".", ",") + "/km";
    }
    function calc() {
      var energia = parseFloat(document.getElementById("calc-energia").value) || 0;
      var consumo = parseFloat(document.getElementById("calc-consumo").value) || 0;
      var gas = parseFloat(document.getElementById("calc-gasolina").value) || 0;
      var moto = parseFloat(document.getElementById("calc-moto").value) || 0;
      outE.textContent = money((energia * consumo) / 100);
      outG.textContent = moto > 0 ? money(gas / moto) : "—";
    }
    ids.forEach(function (id) {
      var el = document.getElementById(id);
      if (el) el.addEventListener("input", calc);
    });
    calc();
  })();
  </script>
