  <script>
  (function () {
    var input = document.getElementById("busca-acessorio");
    var status = document.getElementById("busca-status");
    var cards = Array.prototype.slice.call(document.querySelectorAll(".acessorio"));
    var filters = Array.prototype.slice.call(document.querySelectorAll("#filtros-acessorio [data-filter]"));
    var activeFilter = "all";
    if (!input || !cards.length) return;
    function normalize(s) {
      return String(s || "").toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "");
    }
    function apply() {
      var q = normalize(input.value.trim());
      var shown = 0;
      cards.forEach(function (card) {
        var hay = normalize(card.textContent + " " + (card.getAttribute("data-search") || ""));
        var okText = !q || hay.indexOf(q) > -1;
        var okCat = activeFilter === "all" || card.getAttribute("data-category") === activeFilter;
        var ok = okText && okCat;
        card.style.display = ok ? "" : "none";
        if (ok) shown++;
      });
      if (status) {
        status.textContent = q || activeFilter !== "all"
          ? shown + " item(ns) encontrado(s). Se precisar de algo específico, fale com a ONNE."
          : "Mostrando todos os itens. Digite para filtrar.";
      }
    }
    input.addEventListener("input", apply);
    filters.forEach(function (btn) {
      btn.addEventListener("click", function () {
        activeFilter = btn.getAttribute("data-filter");
        filters.forEach(function (b) {
          b.classList.toggle("pill--solid", b === btn);
        });
        apply();
      });
    });
    var clear = document.getElementById("limpar-busca");
    if (clear) clear.addEventListener("click", function () {
      input.value = "";
      activeFilter = "all";
      filters.forEach(function (b) { b.classList.toggle("pill--solid", b.getAttribute("data-filter") === "all"); });
      apply();
      input.focus();
    });
    apply();
  })();
  </script>
